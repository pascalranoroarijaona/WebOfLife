\begin{abstract}
Discrete Global Grid Systems (DGGS) utilizing spherical icosahedral partitions, such as the H3 hexagonal hierarchy, are foundational for planetary-scale biogeochemical and ecological modeling. In finite-volume formulations on spherical manifolds, adjacent cells exchange mass, momentum, and thermal energy through shared polygonal boundary edges. Due to floating-point truncation, coordinate unwrapping, and icosahedral projection distortions, independent extraction of shared boundary vertices yields spatial dislocations between adjacent cell registries. Left unchecked, these dislocations introduce geometric porosity and non-zero lateral flux divergence across empty space, in direct violation of the First Law of Thermodynamics. This paper presents the mathematical formalization and algorithmic implementation of \texttt{assertBoundaryEndpointTolerance} within \texttt{src/spatial/h3\_adjacency.ts}. Employing a numerically stable Vincenty-Haversine metric on $\mathbb{S}^2$, this mechanism bounds interface porosity to machine precision ($\epsilon_{\text{angular}} \in [10^{-9}, 10^{-6}]$ rad), guaranteeing strict thermodynamic closure and topological manifold consistency.
\end{abstract}

\section{Introduction and Physical Motivation}

In computational ecology and Earth system modeling, continuous conservation laws govern the transport of fluid, carbon, dissolved minerals, and thermal energy:
\begin{equation}
\frac{\partial \rho_X}{\partial t} + \nabla \cdot (\rho_X \mathbf{v}) = \sum \dot{S}_X
\end{equation}
where $\rho_X$ is the volumetric concentration of state variable $X \in \{\text{H}_2\text{O}, \text{DIC}, \text{DO}, \text{Heat}\}$, $\mathbf{v}$ is the advective velocity vector field, and $\dot{S}_X$ denotes local source/sink terms.

When integrating over a discrete global cell $C_u$ with boundary $\partial C_u$, Gauss's Divergence Theorem converts the advective divergence into interface flux integrals:
\begin{equation}
\int_{C_u} \nabla \cdot (\rho_X \mathbf{v}) \, dV = \oint_{\partial C_u} \rho_X \mathbf{v} \cdot \hat{\mathbf{n}} \, dl = \sum_{v \in \mathcal{N}(u)} J_{u \to v}^X
\end{equation}
Conservative transport dictates the fundamental pairwise anti-symmetry condition:
\begin{equation}
J_{u \to v}^X = -J_{v \to u}^X \implies \sum_{u} \sum_{v \in \mathcal{N}(u)} J_{u \to v}^X = 0
\end{equation}

\subsection{The Geometric Dislocation Problem}
In discrete spherical meshes, cell $C_u$ and cell $C_v$ share a directed geodesic arc $E_{uv} = (P_1, P_2)$. However, when extracted independently from cell boundary polygons:
\begin{align}
E_{u} &= (P_{u, 1}, P_{u, 2}) \\
E_{v} &= (P_{v, 2}, P_{v, 1})
\end{align}
Numerical jitter produces an angular displacement $\Delta\sigma = \angle(P_{u,1}, P_{v,2}) > 0$. On a planetary body of mean radius $R_{\oplus} \approx 6.3710088 \times 10^6 \text{ m}$, this angular divergence produces a physical boundary gap:
\begin{equation}
\delta L_{\text{gap}} = R_{\oplus} \cdot \Delta\sigma
\end{equation}
Across an unclosed interface gap $\delta L_{\text{gap}}$, advective leakage occurs at rate:
\begin{equation}
\dot{M}_{\text{leak}}^X = \rho_X \cdot \|\mathbf{v}\| \cdot \delta L_{\text{gap}} \cdot h_{\text{layer}}
\end{equation}
This violates the First Law of Thermodynamics, generating phantom mass and energy accumulation over extended temporal integrations.

\section{Spherical Metric Formulation}

Let points on the two-sphere $\mathbb{S}^2$ be parameterized by latitude $\phi \in [-\frac{\pi}{2}, \frac{\pi}{2}]$ and longitude $\lambda \in [-\pi, \pi]$. Naive evaluation of the central angle using spherical cosine law:
\begin{equation}
\Delta\sigma = \arccos\left(\sin\phi_A \sin\phi_B + \cos\phi_A \cos\phi_B \cos(\lambda_B - \lambda_A)\right)
\end{equation}
suffers from severe catastrophic cancellation when $\Delta\sigma \to 0$.

\subsection{Vincenty-Haversine Hybrid Formulation}
To preserve numerical precision down to $10^{-16}$ rad, we compute:
\begin{align}
\Delta\phi &= \phi_B - \phi_A \\
\Delta\lambda &= \lambda_B - \lambda_A \\
a &= \sin^2\left(\frac{\Delta\phi}{2}\right) + \cos\phi_A \cos\phi_B \sin^2\left(\frac{\Delta\lambda}{2}\right) \\
\Delta\sigma_{AB} &= 2 \arctan2\left(\sqrt{a}, \sqrt{\max(0.0, 1.0 - a)}\right)
\end{align}

Equivalently, converting coordinates to unit Cartesian vectors $\mathbf{u}_A, \mathbf{u}_B \in \mathbb{R}^3$:
\begin{equation}
\mathbf{u}(\phi, \lambda) = \begin{bmatrix} \cos\phi \cos\lambda \\ \cos\phi \sin\lambda \\ \sin\phi \end{bmatrix}, \quad
\Delta\sigma_{AB} = 2 \arcsin\left(\frac{\|\mathbf{u}_A - \mathbf{u}_B\|_2}{2}\right)
\end{equation}

\section{Algorithmic Invariant and Validation Engine}

The topology verification contract is codified as an invariant assertion executed during adjacency graph compilation:

\begin{algorithm}[H]
\caption{Boundary Endpoint Tolerance Invariant Assertion}
\begin{algorithmic}[1]
\REQUIRE Endpoints $P_A, P_B \in \mathbb{S}^2$, maximum angular tolerance $\epsilon_{\text{angular}}$
\STATE $\Delta\sigma_{AB} \leftarrow \text{computeSphericalAngularDistance}(P_A, P_B)$
\IF{$\Delta\sigma_{AB} > \epsilon_{\text{angular}}$}
    \STATE \textbf{throw} \text{BoundaryEndpointToleranceExceededError}($P_A, P_B, \Delta\sigma_{AB}, \epsilon_{\text{angular}}$)
\ENDIF
\RETURN \textbf{true}
\end{algorithmic}
\end{algorithm}

\subsection{Shared Edge Topological Alignment}
For two adjacent cells $C_u$ and $C_v$, the shared edge pair is verified:
\begin{equation}
\text{assertBoundaryEndpointTolerance}(V_{u,1}, V_{v,2}, \epsilon_{\text{angular}})
\end{equation}
\begin{equation}
\text{assertBoundaryEndpointTolerance}(V_{u,2}, V_{v,1}, \epsilon_{\text{angular}})
\end{equation}
Execution halts deterministically before spatial flux monads can register unclosed geometric channels.

\begin{equation}
Q_{uv} = v_{\perp, uv} \cdot A_{uv} = v_{\perp, uv} \cdot (R_{\oplus} \Delta\sigma_{uv}) \cdot h_{\text{layer}}

Upwind state allocation ensures exact pairwise conservation:
\begin{align}
\Delta M_{\text{water}, u} &= -Q_{uv} \cdot \rho_w \cdot \Delta t \\
\Delta M_{\text{water}, v} &= +Q_{uv} \cdot \rho_w \cdot \Delta t \\

\begin{itemize}

    \item Accurate periodic wrap-around across the antimeridian $\lambda = \pm \pi$.
    \item Support for both radian and degree coordinate specifications.
\end{itemize}
