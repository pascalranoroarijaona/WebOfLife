\begin{abstract}
Discrete Global Grid Systems (DGGS) partitioning the spherical manifold $\mathbb{S}^2 \subset \mathbb{R}^3$ into geodesic hexagonal tessellations require consistent directional flux formulations for mass, energy, and trophic transport. In geodesic coordinates, computed boundary normal vectors and flow velocity vectors often suffer from sign arbitrariness caused by vertex winding permutations or tangential projection charts. When a vector opposes the directed displacement vector between adjacent grid centroids, numerical flux calculations invert transport directions, leading to unphysical negative densities and violations of the second law of thermodynamics. We formalize, implement, and verify an isometric orientation projection operator, \texttt{orientVectorTowardsTarget3D}, which evaluates the Euclidean inner product between the candidate vector and the inter-cell displacement vector, inverting the candidate vector sign if and only if the inner product is strictly negative. We establish that this operator is an $L_2$-isometry, preserves zero-flux orthogonality, and guarantees thermodynamic entropy production non-negativity across discrete boundaries.
\end{abstract}

\section{Introduction}
Simulating planetary biogeochemical cycles, atmospheric circulation, and trophic ecological migration requires solving advection-diffusion-reaction equations over the planetary sphere $\mathbb{S}^2$. Discrete Global Grid Systems (DGGS), notably the Uber H3 aperture-7 hexagonal tessellation, provide quasi-uniform spatial discretization, overcoming the singular coordinate poles of classical latitude-longitude grids.

Transport of extensive thermodynamic quantities (e.g., carbon dioxide, moisture, dissolved oxygen, sensible heat, and animal biomass) across the interface $\partial \Omega_{ij}$ connecting cell $C_i$ to cell $C_j$ is governed by boundary normal flux integrations. However, in three-dimensional Cartesian embeddings ($\mathbb{R}^3$), edge-normal vectors and projected velocity vectors are frequently computed via local cross products whose sign depends on arbitrary local vertex orderings.

Failure to resolve directional parity relative to the net displacement $\mathbf{d}_{ij} = \mathbf{p}_j - \mathbf{p}_i$ causes numerical divergence:
\begin{equation}
    \Phi_{ij} = \mathbf{v} \cdot \mathbf{n}_{ij} < 0 \quad \text{when physical transport flows } C_i \to C_j.
\end{equation}
Such parity inversions induce negative stock inventories ($\rho_k < 0$) and reverse entropy generation ($\dot{S} < 0$). This paper establishes the mathematical formulation, algorithmic implementation, and thermodynamic guarantees of the parity orientation operator implemented in the Web of Life engine.

\section{Mathematical Formalism}

\subsection{State Space and Geodesic Displacement}
Let $C_i$ and $C_j$ denote contiguous cells within the DGGS with geodesic centroid coordinates embedded in $\mathbb{R}^3$:
\begin{equation}
    \mathbf{p}_i = \begin{bmatrix} x_i \\ y_i \\ z_i \end{bmatrix}, \quad 
    \mathbf{p}_j = \begin{bmatrix} x_j \\ y_j \\ z_j \end{bmatrix} \in \mathbb{R}^3.
\end{equation}
The directed inter-centroid displacement vector is defined as:
\begin{equation}
    \mathbf{d}_{ij} = \mathbf{p}_j - \mathbf{p}_i = \begin{bmatrix} x_j - x_i \\ y_j - y_i \\ z_j - z_i \end{bmatrix}.
\end{equation}

Let $\mathbf{v} = [v_x, v_y, v_z]^T \in \mathbb{R}^3$ be a candidate advective velocity or face-normal vector associated with the boundary between $C_i$ and $C_j$.

\subsection{Parity Projection Operator}
The standard Euclidean inner product $\langle \mathbf{v}, \mathbf{d}_{ij} \rangle$ is given by:
\begin{equation}
    \langle \mathbf{v}, \mathbf{d}_{ij} \rangle = \mathbf{v} \cdot \mathbf{d}_{ij} = v_x (x_j - x_i) + v_y (y_j - y_i) + v_z (z_j - z_i).
\end{equation}

\begin{definition}[Target Orientation Operator]
The orientation operator $\mathcal{O}: \mathbb{R}^3 \times \mathbb{R}^3 \to \mathbb{R}^3$ maps candidate vector $\mathbf{v}$ and displacement $\mathbf{d}_{ij}$ to oriented vector $\mathbf{v}^*$:
\begin{equation}
    \mathbf{v}^* = \mathcal{O}(\mathbf{v}, \mathbf{d}_{ij}) = 
    \begin{cases}
        -\mathbf{v} = \begin{bmatrix} -v_x \\ -v_y \\ -v_z \end{bmatrix}, & \text{if } \langle \mathbf{v}, \mathbf{d}_{ij} \rangle < 0, \\[10pt]
        \mathbf{v} = \begin{bmatrix} v_x \\ v_y \\ v_z \end{bmatrix}, & \text{if } \langle \mathbf{v}, \mathbf{d}_{ij} \rangle \ge 0.
    \end{cases}
\end{equation}
\end{definition}

\begin{theorem}[Isometric Invariance]
For any $\mathbf{v}, \mathbf{d} \in \mathbb{R}^3$, the Euclidean norm is strictly conserved:
\begin{equation}
    \|\mathcal{O}(\mathbf{v}, \mathbf{d})\|_2 = \|\mathbf{v}\|_2.
\end{equation}
\end{theorem}
\begin{proof}
If $\langle \mathbf{v}, \mathbf{d} \rangle \ge 0$, $\mathbf{v}^* = \mathbf{v}$, and $\|\mathbf{v}^*\|_2 = \|\mathbf{v}\|_2$. If $\langle \mathbf{v}, \mathbf{d} \rangle < 0$, $\mathbf{v}^* = -\mathbf{v}$. By properties of the Euclidean norm on $\mathbb{R}^3$:
\begin{equation}
    \|\mathbf{v}^*\|_2 = \sqrt{(-v_x)^2 + (-v_y)^2 + (-v_z)^2} = \sqrt{v_x^2 + v_y^2 + v_z^2} = \|\mathbf{v}\|_2.
\end{equation}
Hence $\mathcal{O}$ is an exact isometry in $(\mathbb{R}^3, \|\cdot\|_2)$.
\end{proof}

\begin{theorem}[Directional Alignment Semi-Definiteness]
For any non-zero displacement vector $\mathbf{d} \in \mathbb{R}^3 \setminus \{\mathbf{0}\}$, the inner product between the oriented vector $\mathbf{v}^*$ and $\mathbf{d}$ is non-negative:
\begin{equation}
    \langle \mathbf{v}^*, \mathbf{d} \rangle \ge 0.
\end{equation}
\end{theorem}
\begin{proof}
If $\langle \mathbf{v}, \mathbf{d} \rangle \ge 0$, then $\mathbf{v}^* = \mathbf{v}$ and $\langle \mathbf{v}^*, \mathbf{d} \rangle = \langle \mathbf{v}, \mathbf{d} \rangle \ge 0$. If $\langle \mathbf{v}, \mathbf{d} \rangle < 0$, then $\mathbf{v}^* = -\mathbf{v}$ and:
\begin{equation}
    \langle \mathbf{v}^*, \mathbf{d} \rangle = \langle -\mathbf{v}, \mathbf{d} \rangle = - \langle \mathbf{v}, \mathbf{d} \rangle > 0.
\end{equation}
Thus, $\langle \mathbf{v}^*, \mathbf{d} \rangle \ge 0$ holds universally.
\end{proof}

\section{Thermodynamic Transport and Conservation}

\subsection{Mass Exchange Invariance}
Let $A_{ij}$ denote the cross-sectional area of the interface $\partial \Omega_{ij}$, and $\Delta t$ the discrete simulation time step. The effective directed transport velocity is:
\begin{equation}
    u_{\text{eff}} = \frac{\mathbf{v}^* \cdot \mathbf{d}_{ij}}{\|\mathbf{d}_{ij}\|_2} \ge 0.
\end{equation}
The transferred volume delta is $\Delta V_{ij} = u_{\text{eff}} A_{ij} \Delta t$. For state concentration $c_k = S_k(C_i)/V(C_i)$ of conserved physical substance $k$:
\begin{equation}
    \Delta M_k = c_k \Delta V_{ij}, \quad 
    \Delta S_k(C_i) = -\Delta M_k, \quad
    \Delta S_k(C_j) = +\Delta M_k.
\end{equation}
Summing across the directed boundary confirms absolute mass balance:
\begin{equation}
    \Delta S_k(C_i) + \Delta S_k(C_j) \equiv 0.
\end{equation}

\subsection{Entropy Production Non-Negativity}
For sensible enthalpy exchange between thermal reservoirs at temperatures $T_i$ and $T_j$:
\begin{equation}
    \Delta H_{ij} = \rho c_p (T_i - T_{\text{ref}}) \Delta V_{ij}.
\end{equation}
The associated entropy generation rate satisfies:
\begin{equation}
    \Delta S_{\text{universe}} = \Delta H_{ij} \left( \frac{1}{T_j} - \frac{1}{T_i} \right) \ge 0, \quad \forall T_i \ge T_j,
\end{equation}
which is guaranteed because $u_{\text{eff}} \ge 0$ prevents negative advective flux against temperature potentials.

\section{Conclusion}
The implementation of \texttt{orientVectorTowardsTarget3D} provides an essential geometric and thermodynamic invariant operator for discrete global grid systems. By conditioning vector sign on Euclidean displacement parity, the Web of Life engine ensures non-negative concentration dynamics, absolute mass conservation, and thermodynamic stability across global simulation runs.
