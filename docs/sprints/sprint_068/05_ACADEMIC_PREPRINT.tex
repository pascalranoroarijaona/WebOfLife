\title{Deterministic Shared Boundary Geodesic Extraction for Conservative Finite-Volume Flux Transport on Spherical Discrete Global Grids}

\author{Pascal Ranoroarijaona}
\email{dev@weboflife.simulation}

\date{\today}

\begin{abstract}
Planetary-scale geophysical and thermodynamic simulations require strict conservation of mass, momentum, and enthalpy across discrete global control volumes. On spherical geodesic discretizations---such as the icosahedral Snyder aperture-3 hexagonal grid (H3)---isotropic centroid approximations introduce metric truncation errors up to $\sim 15\%$ due to non-uniform edge distortions, directional misalignment, and pentagonal singularities. We formulate and implement \texttt{extractSharedBoundaryVertices3D}, an analytical geometric operator that extracts deterministic three-dimensional Cartesian boundary endpoints $\mathbf{v}_1, \mathbf{v}_2 \in \mathbb{R}^3$ for adjacent control volumes on a spherical manifold $S_R^2$. By establishing exact interface normal vectors $\hat{\mathbf{n}}_{ij}$, geodesic arc lengths $L_{ij}$, and exchange areas $A_{ij}$, this formulation guarantees machine-precision antisymmetry ($\hat{\mathbf{n}}_{ij} \equiv -\hat{\mathbf{n}}_{ji}$) and exact zero-sum First Law conservation for advective-diffusive stock transfers. The algorithm eliminates artificial mass-energy leakages and provides an analytical foundation for high-resolution planetary biogeochemical modeling.
\end{abstract}

\section{Introduction}
Spatial discretization of transport phenomena on spherical planetary bodies is central to atmospheric, oceanic, and biogeochemical modeling. The general governing conservation equation for an extensive stock density $\rho_X$ subject to advection, diffusion, and internal reaction rates $\dot{S}_X$ is given by:
\begin{equation}
\frac{\partial \rho_X}{\partial t} + \nabla \cdot (\mathbf{u} \rho_X) = \nabla \cdot (D_X \nabla \rho_X) + \dot{S}_X
\label{eq:conservation_continuum}
\end{equation}

When integrated over a discrete finite control volume $V_i$, Gauss's Divergence Theorem converts volumetric transport into boundary facet fluxes:
\begin{equation}
\frac{d X_i}{dt} = - \sum_{j \in \mathcal{N}(i)} \int_{\partial V_{ij}} \mathbf{J}_X \cdot \hat{\mathbf{n}}_{ij} \, dA + \int_{V_i} \dot{S}_X \, dV
\label{eq:fvm_integral}
\end{equation}
where $\partial V_{ij}$ denotes the interface between cell $i$ and neighbor $j$, and $\hat{\mathbf{n}}_{ij}$ is the outward unit normal.

In discrete global grid systems (DGGS) based on icosahedral hexagonal tessellations, isotropic approximations often assume uniform edge lengths $\bar{L}$ and approximate the normal vector via the inter-centroid displacement $\mathbf{x}_j - \mathbf{x}_i$. However, hexagonal grid cells projected onto a sphere exhibit edge length variations of up to $15\%$, and contain twelve essential pentagonal singularities per grid level. In asymmetric numerical formulations, $A_{ij} \neq A_{ji}$ or $\hat{\mathbf{n}}_{ij} \neq -\hat{\mathbf{n}}_{ji}$, resulting in spurious mass and energy generation.

\section{Spherical Interface Geometry}

Let $S_R^2 = \{ \mathbf{x} \in \mathbb{R}^3 : \|\mathbf{x}\| = R \}$ denote a planetary sphere of radius $R = 6,371,008\,\text{m}$. A cell $c \in \mathcal{H}$ is defined by an ordered spherical polygon $\mathcal{P}(c) = (\mathbf{u}_0, \mathbf{u}_1, \dots, \mathbf{u}_{k-1})$ with $k \in \{5, 6\}$.

\subsection{Endpoint Extraction Algorithm}
For two topological neighbors $c_A$ and $c_B$, candidate vertices are evaluated under a metric tolerance $\epsilon = 10^{-5} R$:
\begin{equation}
\mathcal{P}(c_A) \cap_{\epsilon} \mathcal{P}(c_B) = \{\mathbf{p}_1, \mathbf{p}_2\}
\end{equation}

To enforce strict antisymmetry, the endpoints are oriented such that the interface tangent $\mathbf{t}_{AB} = \mathbf{p}_2 - \mathbf{p}_1$ generates an outward normal:
\begin{equation}
\mathbf{m} = \mathbf{t}_{AB} \times \mathbf{x}_{c_A}, \quad \hat{\mathbf{n}}_{AB} = \frac{\mathbf{m}}{\|\mathbf{m}\|}
\end{equation}
satisfying the directional orientation criterion:
\begin{equation}
\hat{\mathbf{n}}_{AB} \cdot (\mathbf{x}_{c_B} - \mathbf{x}_{c_A}) > 0
\end{equation}
If the inequality is violated, the vertex order is swapped $[\mathbf{v}_1, \mathbf{v}_2] \leftarrow [\mathbf{p}_2, \mathbf{p}_1]$ and $\hat{\mathbf{n}}_{AB} \leftarrow -\hat{\mathbf{n}}_{AB}$.

\subsection{Analytical Interface Metrics}
With canonical ordering established, the geodesic edge length $L_{AB}$ and interface cross-sectional exchange area $A_{AB}$ for stratum height $\Delta z$ are:
\begin{equation}
L_{AB} = R \arccos\left(\mathrm{clamp}\left(\frac{\mathbf{v}_1 \cdot \mathbf{v}_2}{R^2}, -1.0, 1.0\right)\right)
\end{equation}
\begin{equation}
A_{AB} = L_{AB} \cdot \Delta z
\end{equation}

By algebraic symmetry:
\begin{equation}
L_{AB} \equiv L_{BA}, \quad A_{AB} \equiv A_{BA}, \quad \hat{\mathbf{n}}_{AB} \equiv -\hat{\mathbf{n}}_{BA}
\end{equation}

\section{Conservative Flux Transport}

\subsection{Coupled Advection-Diffusion}
Let $u_n = \mathbf{u}_{ij} \cdot \hat{\mathbf{n}}_{ij}$ denote the normal fluid velocity at the interface midpoint $\mathbf{x}_{ij} = R \frac{\mathbf{v}_1 + \mathbf{v}_2}{\|\mathbf{v}_1 + \mathbf{v}_2\|}$. The total flux density across facet $e_{ij}$ is:
\begin{equation}
J_{X, ij} = u_n \rho_{X, \text{upwind}} - D_X \frac{\rho_{X, j} - \rho_{X, i}}{d_{ij}}
\end{equation}
where $d_{ij} = R \arccos\left(\frac{\mathbf{x}_i \cdot \mathbf{x}_j}{R^2}\right)$, and the upwind density is:
\begin{equation}
\rho_{X, \text{upwind}} = \begin{cases}
\rho_{X, i}, & \text{if } u_n \ge 0 \\
\rho_{X, j}, & \text{if } u_n < 0
\end{cases}
\end{equation}

Discrete stock transfers over timestep $\Delta t$ satisfy:
\begin{equation}
\Delta X_i = - J_{X, ij} A_{ij} \Delta t, \quad \Delta X_j = + J_{X, ij} A_{ij} \Delta t
\end{equation}
giving exact First Law pairwise balance $\Delta X_i + \Delta X_j \equiv 0$.

\subsection{Second Law Thermodynamic Consistency}
Thermal conduction across facet $A_{ij}$ driven by thermal conductivity $k_{\mathrm{th}}$ generates irreversible entropy at the rate:
\begin{equation}
\dot{S}_{\mathrm{gen}, ij} = A_{ij} k_{\mathrm{th}} \frac{(T_i - T_j)^2}{T_i T_j d_{ij}} \ge 0
\end{equation}
Since $A_{ij} > 0$ and $d_{ij} > 0$, non-negative entropy generation is strictly guaranteed for all non-isothermal cell pairs.

\section{Conclusion}
The operator \texttt{extractSharedBoundaryVertices3D} resolves geometric and thermodynamic discrepancies in spherical discrete global grid transport. By extracting deterministic Cartesian endpoints and normal vectors directly on $S_R^2$, the Web of Life simulation architecture ensures rigorous mass and enthalpy conservation across arbitrary pentagonal and hexagonal discrete controls.
