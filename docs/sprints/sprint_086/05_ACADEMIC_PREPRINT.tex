\begin{abstract}
Discrete Global Grid Systems (DGGS) based on icosahedral hexagonal aperture-7 partitioning inherently embed twelve topological pentagonal singularities at their base-cell vertices:
$\mathcal{P}_{\text{base}} = \{4, 14, 24, 38, 49, 58, 63, 72, 83, 97, 107, 117\}$.
Hierarchical aperture-7 decimation algorithms conventionally presuppose a uniform six-neighbor coplanar topology with directional digits $d \in \{0, 1, 2, 3, 4, 5, 6\}$. At pentagonal vertices, the coordinate manifold undergoes topological truncation, suppressing the $K$-axis directional digit ($d=1$) and reducing cell coordination number from 6 to 5. Failure to recognize this singularity introduces artificial divergence into discrete boundary fluxes, violating the First Law of Thermodynamics and injecting negative entropy anomalies into advective-diffusive transport equations. We formalize \texttt{extractPentagonApertureDigits}, a bitwise $\mathcal{O}(r)$ algorithm operating on 64-bit canonical DGGS cell identifiers. This operator decomposes the resolution digit sequence $\mathbf{D} = [d_1, \dots, d_r]$, validates pentagonal invariants, extracts the subsequence of non-zero aperture trajectories $\mathbf{D}_{\neq 0}$, and establishes conductance masking on spatial flux monads. We demonstrate that this formulation maintains machine-precision conservative flux balance ($\sum \Delta M < 10^{-15}\text{ kg}$) and strictly non-negative entropy generation ($\dot{\sigma}_s \ge 0$) over closed spherical geodesics.
\end{abstract}

\section{Introduction}
Global planetary simulation engines require spherical surface partitioning that minimizes metric area and shape distortion. Hexagonal Discrete Global Grid Systems (DGGS) utilizing aperture-7 decimation, notably the Uber H3 framework, achieve uniform spatial discretization by projecting an icosahedron onto the sphere. By Euler's polyhedral formula, a closed 2-manifold of spherical topology ($\chi = 2$) cannot be tessellated exclusively by regular hexagons:
\begin{equation}
V - E + F = 2
\end{equation}
Consequently, exactly twelve base cells situated at the vertices of the icosahedral base mesh project as spherical pentagons rather than hexagons:
\begin{equation}
\mathcal{P}_{\text{base}} = \{4, 14, 24, 38, 49, 58, 63, 72, 83, 97, 107, 117\}
\end{equation}

Each pentagonal base cell possesses 5 immediate neighbors ($N_{\text{pent}} = 5$) and an active surface area:
\begin{equation}
A_{\text{pent}}(r) = \frac{5}{6} A_{\text{hex}}(r) = \frac{5}{6} A_0 \cdot 7^{-r}
\end{equation}

Under aperture-7 hierarchical indexing, every resolution level $k \in [1, r]$ allocates 3 bits to encode a directional aperture digit $d_k \in [0, 6]$. In regular hexagonal cells, $d_k = 0$ specifies the central sub-cell, whereas $d_k \in \{1, 2, 3, 4, 5, 6\}$ identify the six neighboring directional facets along coordinate axes $(I, J, K)$. On pentagonal manifolds, the geometric edge along the $K$-axis ($d=1$) is topologically absent.

If finite-volume differential operators compute fluxes across uniform hexagonal stencils:
\begin{equation}
\nabla \cdot \mathbf{J}_i = \frac{1}{A_i} \sum_{k=1}^6 J_{i \to j}^{(k)} L_k
\end{equation}
any transport attributed to $d=1$ generates non-physical stock divergence, violating the First Law of Thermodynamics.

\section{Mathematical \& Bitwise Specification}

\subsection{Canonical 64-bit Index Architecture}
An H3 index $I$ is structured as a 64-bit unsigned integer:
\begin{equation}
I = \sum_{b=0}^{63} a_b 2^b, \quad a_b \in \{0, 1\}
\end{equation}
Structured bit fields are partitioned as follows:
\begin{itemize}
    \item $[62:59]$: Mode ($1$ for canonical cell mode).
    \item $[55:52]$: Resolution level $r \in [0, 15]$.
    \item $[51:45]$: Base cell identifier $BC \in [0, 121]$.
    \item $[47 - 3k : 45 - 3k]$: Aperture digit $d_k \in [0, 7]$ at resolution $k$.
    \item $[44 - 3r : 0]$: Unused bits, set to binary $1$ ($7$).
\end{itemize}

\subsection{Sequence Extraction Operator}
\begin{definition}[Aperture Sequence Extraction]
Let $\mathcal{E}: \mathbb{N}_{64} \to \mathcal{R}$ be the mapping decomposing index $I$ into its topological resolution components:
\begin{equation}
BC = (I \gg 45) \ \& \ \text{0x7F}
\end{equation}
\begin{equation}
d_k = (I \gg (45 - 3k)) \ \& \ \text{0x07}, \quad \forall k \in \{1, \dots, r\}
\end{equation}
\begin{equation}
\mathbf{D} = [d_1, d_2, \dots, d_r]
\end{equation}
\begin{equation}
\mathbf{D}_{\neq 0} = [d_k \in \mathbf{D} \mid d_k \neq 0]
\end{equation}
\end{definition}

\begin{invariant}[Pure Pentagon Condition]
A cell index $I$ represents a topological pentagon at resolution $r$ if and only if:
\begin{equation}
BC \in \mathcal{P}_{\text{base}} \quad \wedge \quad |\mathbf{D}_{\neq 0}| = 0
\end{equation}
\end{invariant}

\begin{invariant}[Prohibited Aperture Direction]
On any lineage originating from a pentagonal base cell $BC \in \mathcal{P}_{\text{base}}$, the $K$-axis directional digit $d=1$ is invalid:
\begin{equation}
d_k \neq 1, \quad \forall k \in [1, r]
\end{equation}
\end{invariant}

\section{Thermodynamic Flux Conservation}

\subsection{State Vector Representation}
Each cell $i$ tracks a conservative state vector $\mathbf{S}_i \in \mathbb{R}^5$ containing physical stocks of carbon, water, minerals, dissolved oxygen, and internal thermal energy:
\begin{equation}
\mathbf{S}_i = [M_{\text{C}, i}, M_{\text{H}_2\text{O}, i}, M_{\text{min}, i}, M_{\text{O}_2, i}, U_i]^T
\end{equation}

\subsection{Conductance Tensor Topology Masking}
To prevent spurious divergence at pentagonal vertices, the spatial conductance tensor $\mathbf{K}_i \in \mathbb{R}^7$ applies a binary directional validity mask $\mathbf{w}_i(d)$:
\begin{equation}
\mathbf{w}_i(d) = \begin{cases} 
0, & d = 0 \\
0, & d = 1 \text{ and } BC_i \in \mathcal{P}_{\text{base}} \\
1, & d \in \{2, 3, 4, 5, 6\} \text{ and } BC_i \in \mathcal{P}_{\text{base}} \\
1, & d \in \{1, 2, 3, 4, 5, 6\} \text{ and } BC_i \notin \mathcal{P}_{\text{base}}
\end{cases}
\end{equation}

The effective directional conductance is defined by:
\begin{equation}
\mathbf{K}_i(d) = \mathbf{w}_i(d) \cdot K_{\text{nominal}} \cdot \alpha_{\text{geom}}(i)
\end{equation}
where $\alpha_{\text{geom}}(i) = \frac{5}{6}$ if $BC_i \in \mathcal{P}_{\text{base}} \wedge |\mathbf{D}_{\neq 0}| = 0$, and $1.0$ otherwise.

\begin{theorem}[Global Mass-Energy Conservation]
For a closed control volume $\Omega$ encompassing pentagonal cells, inter-cell transfers satisfy:
\begin{equation}
\sum_{i \in \Omega} \Delta \mathbf{S}_i = \mathbf{0}
\end{equation}
\end{theorem}
\begin{proof}
Across each interface between adjacent cells $i$ and $j$ along valid facet direction $d$, the fluxes satisfy anti-symmetry by construction:
\begin{equation}
\mathbf{J}_{i \to j}^{(d)} = - \mathbf{J}_{j \to i}^{(d')}
\end{equation}
For facet $d=1$ adjacent to a pentagonal cell, $\mathbf{w}_i(1) = 0 \implies \mathbf{K}_i(1) = 0$, yielding identically zero flux:
\begin{equation}
\mathbf{J}_{i \to j}^{(1)} \equiv \mathbf{0}
\end{equation}
Summing all boundary interfaces yields exact pairwise cancellation.
\end{proof}

\subsection{Second Law Entropy Dissipation}
Local entropy production $\dot{\sigma}_{s, ij}^{(k)}$ along facet $k$ is given by:
\begin{equation}
\dot{\sigma}_{s, ij}^{(k)} = J_{U, i \to j}^{(k)} \left( \frac{1}{T_j} - \frac{1}{T_i} \right) + \sum_{s} J_{s, i \to j}^{(k)} \left( \frac{\mu_{s, i}}{T_i} - \frac{\mu_{s, j}}{T_j} \right) \ge 0
\end{equation}
Suppressing facet $d=1$ eliminates coordinate singularity artifacts that could otherwise produce negative entropy generation rates ($\dot{\sigma}_s < 0$).

\section{Implementation in TypeScript}
The extraction algorithm was implemented in \texttt{src/spatial/h3\_adjacency.ts}:
\begin{lstlisting}[language=Java]
export function extractPentagonApertureDigits(
  h3IndexHex: string
): PentagonApertureResult {
  const val = BigInt("0x" + h3IndexHex.trim());
  const mode = Number((val >> 59n) & 0xfn);
  if (mode !== 1) throw new Error("Invalid mode");
  
  const resolution = Number((val >> 52n) & 0xfn);
  const baseCell = Number((val >> 45n) & 0x7fn);
  const isPentagonBaseCell = PENTAGONS.has(baseCell);
  
  const allDigits: number[] = [];
  const nonZeroDigits: number[] = [];
  let leadingNonZeroDigit: number | null = null;
  let leadingNonZeroResolution: number | null = null;
  let leadingCenterCount = 0;
  let hasInvalidPentagonDigit = false;
  let encounteredNonZero = false;

  for (let k = 1; k <= resolution; k++) {
    const shift = 45n - 3n * BigInt(k);
    const digit = Number((val >> shift) & 7n);
    allDigits.push(digit);

    if (digit === 0) {
      if (!encounteredNonZero) leadingCenterCount++;
    } else {
      if (!encounteredNonZero) {
        leadingNonZeroDigit = digit;
        leadingNonZeroResolution = k;
        encounteredNonZero = true;
      }
      nonZeroDigits.push(digit);
    }
    if (isPentagonBaseCell && digit === 1) {
      hasInvalidPentagonDigit = true;
    }
  }

  return Object.freeze({
    h3Index: h3IndexHex.toLowerCase(),
    resolution,
    baseCell,
    isPentagonBaseCell,
    isPurePentagon: isPentagonBaseCell && nonZeroDigits.length === 0,
    allDigits: Object.freeze(allDigits),
    nonZeroDigits: Object.freeze(nonZeroDigits),
    leadingNonZeroDigit,
    leadingNonZeroResolution,
    leadingCenterCount,
    hasInvalidPentagonDigit
  });
}
\end{lstlisting}
